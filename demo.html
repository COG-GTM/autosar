Application Primitive Data Type
uint8 with data constraints (0-255)
▾
Python Source application_data_type.py 
import autosar.xml
import autosar.xml.element as ar_element

workspace = autosar.xml.Workspace()
workspace.create_package_map({
    "ApplicationDataTypes": "DataTypes/ApplicationDataTypes",
    "DataConstrs": "DataTypes/DataConstrs"
})

# Create a data constraint (0 to 255)
data_constraint = ar_element.DataConstraint.make_internal(
    "uint8_ADT_DataConstr", 0, 255)
workspace.add_element("DataConstrs", data_constraint)

# Create the application data type
sw_data_def_props = ar_element.SwDataDefPropsConditional(
    data_constraint_ref=data_constraint.ref())
data_type = ar_element.ApplicationPrimitiveDataType(
    "uint8_ADT", category="VALUE",
    sw_data_def_props=sw_data_def_props)
workspace.add_element("ApplicationDataTypes", data_type)

workspace.create_document("datatypes.arxml", packages="/DataTypes")
workspace.write_documents()
Generated ARXML datatypes.arxml 
<AR-PACKAGE>
  <SHORT-NAME>ApplicationDataTypes</SHORT-NAME>
  <ELEMENTS>
    <APPLICATION-PRIMITIVE-DATA-TYPE>
      <SHORT-NAME>uint8_ADT</SHORT-NAME>
      <CATEGORY>VALUE</CATEGORY>
      <SW-DATA-DEF-PROPS>
        <SW-DATA-DEF-PROPS-CONDITIONAL>
          <DATA-CONSTR-REF DEST="DATA-CONSTR">
            /DataTypes/DataConstrs/uint8_ADT_DataConstr
          </DATA-CONSTR-REF>
        </SW-DATA-DEF-PROPS-CONDITIONAL>
      </SW-DATA-DEF-PROPS>
    </APPLICATION-PRIMITIVE-DATA-TYPE>
  </ELEMENTS>
</AR-PACKAGE>

Python Source compu_method_rational.py 
package = ar_element.Package("CompuMethods")
computation = ar_element.Computation.make_rational(
    scaling_factor=1/64,
    offset=0,
    lower_limit=0,
    upper_limit=65535,
    default_value=65535)
package.append(ar_element.CompuMethod(
    name='VehicleSpeed_T',
    int_to_phys=computation,
    category="LINEAR"))

# Write and read back

writer.write_file(document, file_path)
reader = autosar.xml.Reader()

# Create the SWC
swc = ar_element.ApplicationSoftwareComponentType(
    "SenderComponent")
workspace.add_element("ComponentTypes", swc)

# Create provide ports with init values
swc.create_provide_port("EngineSpeed",
    engine_speed_interface,
    com_spec={"init_value": engine_speed_init.ref(),
              "uses_end_to_end_protection": False})

# Create internal behavior
behavior = swc.create_internal_behavior()
behavior.create_exclusive_area("ExampleExclusiveArea")

# Create runnables
behavior.create_runnable("SenderComponent_Init")
runnable = behavior.create_runnable("SenderComponent_Run",
    can_enter_leave="ExampleExclusiveArea")
runnable.create_port_access(
    ["EngineSpeed", "VehicleSpeed"])

# Create events

behavior.create_timing_event("SenderComponent_Run",
    period=0.1)

# Port API options
